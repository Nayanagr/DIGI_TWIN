### 2. The Logic (`app.js`)

This script handles the viewport sizing, captures your mouse clicks to calculate grid coordinates, and fires the payload to your FastAPI backend. When the backend returns the processed 2D matrix, this script paints the heatmap over your city.

```javascript
// --- DOM Elements ---
const mapContainer = document.getElementById('map-container');
const heatmapCanvas = document.getElementById('heatmap-canvas');
const uiCanvas = document.getElementById('ui-canvas');
const heatCtx = heatmapCanvas.getContext('2d');
const uiCtx = uiCanvas.getContext('2d');

const btnSimulate = document.getElementById('btn-simulate');
const btnClear = document.getElementById('btn-clear');
const coordReadout = document.getElementById('coord-readout');
const sourceCountText = document.getElementById('source-count');

const windSpeedInput = document.getElementById('wind-speed');
const windSpeedVal = document.getElementById('ws-val');
const windDirInput = document.getElementById('wind-dir');
const windDirVal = document.getElementById('wd-val');
const stabilityInput = document.getElementById('stability-class');

// --- Global State ---
let sources = [];
const GRID_RESOLUTION = 100; // Must match backend assumption

// --- Initialization ---
function resizeCanvases() {
    heatmapCanvas.width = window.innerWidth;
    heatmapCanvas.height = window.innerHeight;
    uiCanvas.width = window.innerWidth;
    uiCanvas.height = window.innerHeight;
    drawUI(); // Redraw crosshairs if window resizes
}

window.addEventListener('resize', resizeCanvases);
resizeCanvases();

// --- UI Listeners ---
windSpeedInput.addEventListener('input', (e) => windSpeedVal.innerText = e.target.value);
windDirInput.addEventListener('input', (e) => windDirVal.innerText = e.target.value);

// --- Mouse Tracking & Coordinate Mapping ---
mapContainer.addEventListener('mousemove', (e) => {
    // Map raw screen pixels to our 100x100 simulation grid
    const gridX = Math.floor((e.clientX / window.innerWidth) * GRID_RESOLUTION);
    const gridY = Math.floor((1 - (e.clientY / window.innerHeight)) * GRID_RESOLUTION); // Invert Y so 0 is bottom
    coordReadout.innerText = `GRID TGT: [X: ${gridX}, Y: ${gridY}]`;
});

mapContainer.addEventListener('click', (e) => {
    // Add a source on click
    const gridX = Math.floor((e.clientX / window.innerWidth) * GRID_RESOLUTION);
    const gridY = Math.floor((1 - (e.clientY / window.innerHeight)) * GRID_RESOLUTION);

    sources.push({
        x: gridX,
        y: gridY,
        rate: 50.0,          // Default emission rate
        stack_height: 50.0   // Default stack height in meters
    });

    sourceCountText.innerText = sources.length;
    drawUI();
});

function drawUI() {
    uiCtx.clearRect(0, 0, uiCanvas.width, uiCanvas.height);
    
    sources.forEach(src => {
        // Map grid coordinates back to screen pixels
        const screenX = (src.x / GRID_RESOLUTION) * window.innerWidth;
        const screenY = window.innerHeight - ((src.y / GRID_RESOLUTION) * window.innerHeight);

        // Draw tactical crosshair
        uiCtx.beginPath();
        uiCtx.arc(screenX, screenY, 6, 0, 2 * Math.PI);
        uiCtx.strokeStyle = '#0f0';
        uiCtx.lineWidth = 2;
        uiCtx.stroke();
        
        uiCtx.beginPath();
        uiCtx.moveTo(screenX - 10, screenY);
        uiCtx.lineTo(screenX + 10, screenY);
        uiCtx.moveTo(screenX, screenY - 10);
        uiCtx.lineTo(screenX, screenY + 10);
        uiCtx.stroke();
    });
}

// --- API Communication & Rendering ---
btnSimulate.addEventListener('click', async () => {
    if (sources.length === 0) {
        alert("COMMAND REJECTED: Place at least one emission source.");
        return;
    }

    btnSimulate.innerText = "CALCULATING...";
    btnSimulate.style.background = "#f0ad4e";

    const payload = {
        grid_size: GRID_RESOLUTION,
        wind_speed: parseFloat(windSpeedInput.value),
        wind_direction_deg: parseFloat(windDirInput.value),
        stability_class: stabilityInput.value,
        sources: sources
    };

    try {
        // Ensure your FastAPI backend is running on port 8000
        const response = await fetch('http://localhost:8000/simulate', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });

        if (!response.ok) throw new Error("Backend connection failed.");

        const data = await response.json();
        renderHeatmap(data.grid);

    } catch (error) {
        console.error(error);
        alert("CONNECTION FAILED. Is the Python FastAPI engine running?");
    } finally {
        btnSimulate.innerText = "EXECUTE DISPERSION";
        btnSimulate.style.background = "#d9534f";
    }
});

function renderHeatmap(gridArray) {
    heatCtx.clearRect(0, 0, heatmapCanvas.width, heatmapCanvas.height);
    
    const cellWidth = heatmapCanvas.width / GRID_RESOLUTION;
    const cellHeight = heatmapCanvas.height / GRID_RESOLUTION;

    for (let y = 0; y < GRID_RESOLUTION; y++) {
        for (let x = 0; x < GRID_RESOLUTION; x++) {
            const val = gridArray[y][x];
            if (val > 1) { // Threshold to ignore near-zero noise
                // Map the 0-255 value to a cold-to-hot color scale
                const r = val;
                const g = 255 - val;
                const b = 0;
                const alpha = (val / 255) * 0.8; // Higher concentration = more opaque
                
                heatCtx.fillStyle = `rgba(${r}, ${g}, ${b}, ${alpha})`;
                // Remember backend Y=0 is bottom, Canvas Y=0 is top
                const screenY = heatmapCanvas.height - ((y + 1) * cellHeight);
                heatCtx.fillRect(x * cellWidth, screenY, cellWidth + 1, cellHeight + 1);
            }
        }
    }
}

btnClear.addEventListener('click', () => {
    sources = [];
    sourceCountText.innerText = "0";
    heatCtx.clearRect(0, 0, heatmapCanvas.width, heatmapCanvas.height);
    uiCtx.clearRect(0, 0, uiCanvas.width, uiCanvas.height);
});

```

### 3. Execution Protocol

You must run the backend and the frontend simultaneously.

1. **Boot the Engine (Backend):** Open a terminal, navigate to your backend folder where your Python files are, and start the FastAPI server:
```bash
uvicorn main:app --reload --port 8000

```


2. **Boot the HUD (Frontend):** Open a *second* terminal, navigate to your `frontend` folder, and spin up a lightweight HTTP server:
```bash
python -m http.server 3000

```


3. **Engage:** Open your browser and go to `http://localhost:3000`. Click the map to place targets, adjust the wind, and hit Execute.

Break the system down. Report back when the grid is operational.
